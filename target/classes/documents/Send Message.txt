title Send Message

actor Client/Driver
participant MessagingPage
participant MessageApiClass
participant RestApiController
database Database
participant SocketClient
participant SocketServer

Client/Driver->MessagingPage: Add input
Client/Driver->MessagingPage: Click send button
activate MessagingPage
  MessagingPage->MessageApiClass: Send message data
deactivate MessagingPage

activate MessageApiClass
  MessageApiClass->RestApiController: Send PUT request with data
deactivate MessageApiClass

activate RestApiController
  RestApiController->Database: Add new message
  activate Database
    Database->Database: Create/Update new message document
    Database->RestApiController: Send response
  deactivate Database

  alt Message document is created
    RestApiController->MessageApiClass:Send response
    activate MessageApiClass
      MessageApiClass->SocketClient: Send recipient ID
    deactivate MessageApiClass
    activate SocketClient
      SocketClient->SocketServer:Send recipient ID
      activate SocketServer
        SocketServer->SocketServer: Notifiy recipient
        SocketServer->SocketClient:Send response
      deactivate SocketServer
      SocketClient->MessagingPage:Send successful response
    deactivate SocketClient
    activate MessagingPage
      MessagingPage->Client/Driver:Add message into chatbox
    deactivate MessagingPage
  else
    RestApiController->Client/Driver: Display error message
  end

deactivate RestApiController
